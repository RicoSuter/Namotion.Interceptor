```csharp
using System.Device.Gpio;
using Namotion.Interceptor;

namespace Namotion.Devices.Gpio;

/// <summary>
/// Exclusive use of GPIO pins of a <see cref="GpioSubject"/>, created by <see cref="GpioSubject.LeasePins"/>.
/// Disposing it returns the pins to the subject.
/// </summary>
public sealed class GpioPinLease : IDisposable
{
    private readonly GpioSubject _subject;
    private readonly CancellationTokenSource _revoked = new();

    internal GpioPinLease(GpioSubject subject, IInterceptorSubject owner, IReadOnlyDictionary<int, GpioPinLeaseMode> modes)
    {
        _subject = subject;
        Owner = owner;
        Modes = modes;
    }

    internal IInterceptorSubject Owner { get; }

    internal IReadOnlyDictionary<int, GpioPinLeaseMode> Modes { get; }

    /// <summary>
    /// Canceled when the lease ends: when it is disposed, or when the subject releases its controller because it stops,
    /// is disposed or switches between simulation and hardware. A reload that keeps the controller keeps the lease.
    /// Callbacks run on the thread pool.
    /// </summary>
    public CancellationToken Revoked => _revoked.Token;

    /// <summary>Drives a pin leased as <see cref="GpioPinLeaseMode.Output"/>.</summary>
    /// <exception cref="ArgumentException">The pin is not part of the lease.</exception>
    /// <exception cref="InvalidOperationException">The pin is not an output, or the lease ended.</exception>
    public void Write(int pinNumber, PinValue value) => _subject.WriteLeasedPin(this, pinNumber, value);

    /// <summary>Reads a pin leased as an input or an output.</summary>
    /// <exception cref="ArgumentException">The pin is not part of the lease.</exception>
    /// <exception cref="InvalidOperationException">The pin is <see cref="GpioPinLeaseMode.Reserved"/>, or the lease ended.</exception>
    public PinValue Read(int pinNumber) => _subject.ReadLeasedPin(this, pinNumber);

    /// <summary>Ends the lease and returns the pins to the subject, which opens them in their configured mode.</summary>
    public void Dispose() => _subject.ReleaseLease(this);

    // Callbacks of the token run on the thread pool, so that none runs under the controller lock of the subject.
    internal void MarkRevoked() => _ = _revoked.CancelAsync();
}
```

```csharp
using System.ComponentModel;
using System.Device.Gpio;
using System.Reflection;
using HomeBlaze.Abstractions;
using Namotion.Interceptor;

namespace Namotion.Devices.Gpio;

public partial class GpioSubject
{
    // Guarded by _controllerLock. Every lease holds an open controller: Stop revokes them before it releases it.
    private readonly List<GpioPinLease> _leases = [];

    /// <summary>
    /// Leases pins for exclusive use by <paramref name="owner"/>. The pins stay in <see cref="Pins"/> but are detached,
    /// show <see cref="ServiceStatus.Unavailable"/> with the message <c>Used by &lt;owner&gt;</c>, and are opened in the
    /// requested mode, except <see cref="GpioPinLeaseMode.Reserved"/> pins, which are closed and not opened.
    /// </summary>
    /// <param name="owner">The subject that uses the pins. Its <see cref="ITitleProvider.Title"/>, or else the display
    /// name of its type, names it in the message of the pins.</param>
    /// <param name="pins">The pins to lease.</param>
    /// <returns>The lease, which returns the pins when disposed.</returns>
    /// <exception cref="InvalidOperationException">The subject is not running, a pin is reserved for an ADC or already
    /// leased, or the owner already holds a lease.</exception>
    /// <exception cref="ArgumentOutOfRangeException">A pin number is outside the pins of the controller.</exception>
    /// <exception cref="ArgumentException">A pin number is requested twice.</exception>
    public GpioPinLease LeasePins(IInterceptorSubject owner, IReadOnlyList<GpioPinLeaseRequest> pins)
    {
        ArgumentNullException.ThrowIfNull(owner);
        ArgumentNullException.ThrowIfNull(pins);

        lock (_controllerLock)
        {
            if (_controller is null || Status != ServiceStatus.Running)
            {
                throw new InvalidOperationException("GPIO is not running.");
            }

            var modes = ValidateLeaseRequest(owner, pins);
            var controller = _controller;
            var failureMessages = new List<string>();
            foreach (var pinNumber in modes.Keys)
            {
                if (_attachedPins.Remove(pinNumber, out var attachedPin))
                {
                    attachedPin.Detach();
                    UnregisterInterrupt(pinNumber);
                }

                ClosePin(pinNumber, failureMessages);
            }

            var openedPinNumbers = new List<int>();
            try
            {
                foreach (var (pinNumber, mode) in modes)
                {
                    if (mode == GpioPinLeaseMode.Reserved)
                    {
                        continue;
                    }

                    if (mode == GpioPinLeaseMode.Output)
                    {
                        controller.OpenPin(pinNumber, PinMode.Output, PinValue.Low);
                    }
                    else
                    {
                        controller.OpenPin(pinNumber, ToPinMode(mode));
                    }

                    openedPinNumbers.Add(pinNumber);
                }
            }
            catch
            {
                foreach (var pinNumber in openedPinNumbers)
                {
                    ClosePin(pinNumber, failureMessages);
                }

                AttachPins(modes.Keys);
                throw;
            }

            var message = $"Used by {GetOwnerName(owner)}";
            foreach (var pinNumber in modes.Keys)
            {
                if (Pins.TryGetValue(pinNumber, out var pin))
                {
                    pin.StatusMessage = message;
                    pin.Status = ServiceStatus.Unavailable;
                }
            }

            var lease = new GpioPinLease(this, owner, modes);
            _leases.Add(lease);
            return lease;
        }
    }

    internal void ReleaseLease(GpioPinLease lease)
    {
        lock (_controllerLock)
        {
            if (_leases.Remove(lease))
            {
                CloseLeasedPins(lease, []);
                AttachPins(lease.Modes.Keys);
            }
        }

        lease.MarkRevoked();
    }

    internal void WriteLeasedPin(GpioPinLease lease, int pinNumber, PinValue value)
    {
        lock (_controllerLock)
        {
            if (GetLeasedMode(lease, pinNumber) != GpioPinLeaseMode.Output)
            {
                throw new InvalidOperationException($"GPIO pin {pinNumber} is not leased as an output.");
            }

            _controller!.Write(pinNumber, value);
        }
    }

    internal PinValue ReadLeasedPin(GpioPinLease lease, int pinNumber)
    {
        lock (_controllerLock)
        {
            if (GetLeasedMode(lease, pinNumber) == GpioPinLeaseMode.Reserved)
            {
                throw new InvalidOperationException($"GPIO pin {pinNumber} is reserved and not open.");
            }

            return _controller!.Read(pinNumber);
        }
    }

    // Called under _controllerLock.
    private string? GetLeasedPinMessage(int pinNumber)
    {
        foreach (var lease in _leases)
        {
            if (lease.Modes.ContainsKey(pinNumber))
            {
                return $"Used by {GetOwnerName(lease.Owner)}";
            }
        }

        return null;
    }

    // Called under _controllerLock before the controller is released. With an injected driver the controller is only
    // dropped, so the pins of a lease are closed here or they would stay open in the driver.
    private void RevokeLeases(List<string> failureMessages)
    {
        foreach (var lease in _leases)
        {
            CloseLeasedPins(lease, failureMessages);
            lease.MarkRevoked();
        }

        _leases.Clear();
    }

    // Called under _controllerLock.
    private Dictionary<int, GpioPinLeaseMode> ValidateLeaseRequest(IInterceptorSubject owner, IReadOnlyList<GpioPinLeaseRequest> pins)
    {
        if (_leases.Exists(lease => ReferenceEquals(lease.Owner, owner)))
        {
            throw new InvalidOperationException($"{GetOwnerName(owner)} already holds a GPIO pin lease.");
        }

        var isI2cBus1Reserved = IsI2cBus1Reserved();
        var modes = new Dictionary<int, GpioPinLeaseMode>(pins.Count);
        foreach (var request in pins)
        {
            if (request.PinNumber < 0 || request.PinNumber >= _pinCount)
            {
                throw new ArgumentOutOfRangeException(nameof(pins), request.PinNumber, $"GPIO pin {request.PinNumber} does not exist.");
            }

            if (!modes.TryAdd(request.PinNumber, request.Mode))
            {
                throw new ArgumentException($"GPIO pin {request.PinNumber} is requested twice.", nameof(pins));
            }

            if ((GetLeasedPinMessage(request.PinNumber) ?? GetReservedPinMessage(request.PinNumber, Mcp3008, isI2cBus1Reserved)) is { } message)
            {
                throw new InvalidOperationException($"GPIO pin {request.PinNumber} cannot be leased: {message}.");
            }
        }

        return modes;
    }

    // Called under _controllerLock.
    private GpioPinLeaseMode GetLeasedMode(GpioPinLease lease, int pinNumber)
    {
        if (!_leases.Contains(lease))
        {
            throw new InvalidOperationException("The GPIO pin lease has ended.");
        }

        return lease.Modes.TryGetValue(pinNumber, out var mode)
            ? mode
            : throw new ArgumentException($"GPIO pin {pinNumber} is not part of the lease.", nameof(pinNumber));
    }

    // Called under _controllerLock with the controller.
    private void CloseLeasedPins(GpioPinLease lease, List<string> failureMessages)
    {
        foreach (var (pinNumber, mode) in lease.Modes)
        {
            if (mode != GpioPinLeaseMode.Reserved)
            {
                ClosePin(pinNumber, failureMessages);
            }
        }
    }

    // Called under _controllerLock with the controller: attaches the given pins again, as Connect does.
    private void AttachPins(IEnumerable<int> pinNumbers)
    {
        foreach (var pinNumber in pinNumbers)
        {
            if (!Pins.TryGetValue(pinNumber, out var pin))
            {
                continue;
            }

            try
            {
                _attachedPins[pinNumber] = pin;
                pin.Attach(_controller!, _registerInterrupt, _unregisterInterrupt);
            }
            catch (Exception exception)
            {
                TrySetError(pin, $"GPIO pin {pinNumber} could not be started: {exception.Message}");
            }
        }
    }

    private static string GetOwnerName(IInterceptorSubject owner) =>
        owner is ITitleProvider { Title: { Length: > 0 } title }
            ? title
            : owner.GetType().GetCustomAttribute<DisplayNameAttribute>()?.DisplayName ?? owner.GetType().Name;

    private static PinMode ToPinMode(GpioPinLeaseMode mode) => mode switch
    {
        GpioPinLeaseMode.InputPullUp => PinMode.InputPullUp,
        GpioPinLeaseMode.InputPullDown => PinMode.InputPullDown,
        GpioPinLeaseMode.Output => PinMode.Output,
        _ => PinMode.Input
    };
}
```

```csharp
namespace HomeBlaze.Abstractions;

/// <summary>
/// Parses texts with expressions between double braces, such as <c>Charging {{ Wallbox/ChargingPower }}</c>, as the
/// host evaluates and formats them. A subject gets it from its context with <c>TryGetService</c>.
/// </summary>
public interface ITextTemplateRenderer
{
    /// <summary>Parses a text once, so that it can be rendered many times.</summary>
    /// <param name="text">The text as written.</param>
    /// <param name="template">The parsed template, or null when the text has a mistake.</param>
    /// <param name="error">What is wrong with the text, such as <c>Unknown pipe 'rounded'</c>, or null.</param>
    /// <returns>Whether the text was parsed.</returns>
    bool TryParse(string text, [NotNullWhen(true)] out ITextTemplate? template, [NotNullWhen(false)] out string? error);
}

/// <summary>
/// A text parsed by <see cref="ITextTemplateRenderer.TryParse"/>. Thread-safe.
/// </summary>
public interface ITextTemplate
{
    /// <summary>The text as written.</summary>
    string Text { get; }

    /// <summary>
    /// Renders the text with the current values. Paths resolve from <paramref name="scope"/> as the host defines it.
    /// </summary>
    /// <param name="scope">The subject the paths resolve from, such as the subject that holds the template.</param>
    /// <param name="encode">Applied to the text of each evaluated expression, never to the literal text around it,
    /// such as an XML encoder for a template that is markup, or null to insert the values as they are.</param>
    string Render(IInterceptorSubject scope, Func<string, string>? encode = null);
```

```csharp
using System.Collections.Immutable;
using System.Diagnostics.CodeAnalysis;
using HomeBlaze.Abstractions;
using Namotion.Interceptor;
using Namotion.Interceptor.Expressions;
using Namotion.Interceptor.Registry.Abstractions;

namespace HomeBlaze.Services.Display;

/// <summary>
/// Renders text templates as widget texts are rendered: paths resolve from the scope of the subject, or from the root
/// with <c>/</c>, and values are formatted with their units, as the pipes and value maps of pages format them.
/// </summary>
internal sealed class SubjectTextTemplateRenderer(SubjectPathResolver resolver) : ITextTemplateRenderer
{
    public bool TryParse(string text, [NotNullWhen(true)] out ITextTemplate? template, [NotNullWhen(false)] out string? error)
    {
        ArgumentNullException.ThrowIfNull(text);
        if (!TextTemplate.TryParse(text, ExpressionDisplay.PathSeparator, out var parsed, out error))
        {
            template = null;
            return false;
        }

        foreach (var expression in parsed.Expressions)
        {
            if (ExpressionDisplay.Check(expression, ExpressionUse.TemplateText) is { } checkError)
            {
                template = null;
                error = checkError;
                return false;
            }
        }

        template = new SubjectTextTemplate(resolver, parsed);
        return true;
    }

    private sealed class SubjectTextTemplate(SubjectPathResolver resolver, TextTemplate template) : ITextTemplate
    {
        public string Text => template.Text;

        public string Render(IInterceptorSubject scope, Func<string, string>? encode = null)
        {
            ArgumentNullException.ThrowIfNull(scope);
            var context = new TemplateExpressionContext(resolver, scope, encode);
            return template.Render(ref context);
        }
    }

    private readonly struct TemplateExpressionContext(
        SubjectPathResolver resolver,
        IInterceptorSubject scope,
        Func<string, string>? encode) : IExpressionContext
    {
        public ExpressionValue ResolvePath(string path)
        {
            RegisteredSubjectProperty? property;
            try
            {
                property = resolver.ResolveScopedDisplayableProperty(path, PathStyle.Canonical, scope);
            }
            catch
            {
                // The path is written by the author of the template, and a malformed one must not stop the render.
                return ExpressionValue.Missing;
            }

            return property is null ? ExpressionValue.Missing : ExpressionValue.FromProperty(property, property.GetValue());
        }

        // A text template binds no variable, and TryParse refuses one.
        public ExpressionValue ResolveVariable(string variable, string? memberPath) => ExpressionValue.Missing;

        public ExpressionValue ApplyPipe(string name, ImmutableArray<string> arguments, ExpressionValue input) =>
            ExpressionDisplay.ApplyPipe(name, arguments, input, resolver, scope, timeZone: null);

        public string ToText(ExpressionValue result)
        {
            var text = ExpressionDisplay.ToText(result, timeZone: null);
            return encode is null ? text : encode(text);
        }
    }
}
```

```csharp
/// An immutable black and white image with one bit per pixel: rows padded to whole bytes, the most significant bit

/// </summary>

    private const int LuminanceThreshold = 128;

    private readonly byte[] _data;

    private MonochromeFrame(int width, int height, byte[] data)
    {
        Width = width;
        Height = height;
        _data = data;
    }

    /// <summary>The width in pixels.</summary>
    public int Width { get; }

    /// <summary>The height in pixels.</summary>
    public int Height { get; }

    /// <summary>The bytes of a row: the width rounded up to whole bytes.</summary>
    public int BytesPerRow => (Width + 7) / 8;

    /// <summary>The packed pixels, row by row.</summary>
    public ReadOnlyMemory<byte> Data => _data;

    /// <summary>
    /// Converts a bitmap by a luminance threshold of 50 %, which also turns antialiased edges into whole pixels.
    /// </summary>
    public static MonochromeFrame FromBitmap(SKBitmap bitmap)
    {
        ArgumentNullException.ThrowIfNull(bitmap);
        var frame = CreateWhite(bitmap.Width, bitmap.Height);
        for (var y = 0; y < bitmap.Height; y++)
        {
            for (var x = 0; x < bitmap.Width; x++)
            {
                var color = bitmap.GetPixel(x, y);
                // Transparent pixels show the white of the panel.
                var alpha = color.Alpha / 255.0;
                var luminance = (0.299 * color.Red + 0.587 * color.Green + 0.114 * color.Blue) * alpha + 255 * (1 - alpha);
                if (luminance < LuminanceThreshold)
                {
                    frame.SetBlack(x, y);
                }
            }
        }

        return frame;
    }

    /// <summary>Whether the pixel is white.</summary>
    public bool IsWhite(int x, int y) => (_data[y * BytesPerRow + x / 8] & (0x80 >> (x % 8))) != 0;

    /// <summary>
    /// Rotates the image counter-clockwise by 0, 90, 180 or 270 degrees.
    /// </summary>
    /// <exception cref="ArgumentOutOfRangeException"><paramref name="degrees"/> is not a quarter turn.</exception>
    public MonochromeFrame Rotate(int degrees)
    {
        if (degrees is not (0 or 90 or 180 or 270))
        {
            throw new ArgumentOutOfRangeException(nameof(degrees), degrees, "The rotation must be 0, 90, 180 or 270 degrees.");
        }

        if (degrees == 0)
        {
            return this;
        }

        var isQuarterTurn = degrees != 180;
        var rotated = CreateWhite(isQuarterTurn ? Height : Width, isQuarterTurn ? Width : Height);
        for (var y = 0; y < Height; y++)
        {
            for (var x = 0; x < Width; x++)
            {
                if (IsWhite(x, y))
                {
                    continue;
                }

                var (targetX, targetY) = degrees switch
                {
                    90 => (y, Width - 1 - x),
                    180 => (Width - 1 - x, Height - 1 - y),
                    _ => (Height - 1 - y, x)
                };
                rotated.SetBlack(targetX, targetY);
            }
        }

        return rotated;
    }

    /// <summary>Encodes the frame as a PNG image.</summary>
    public byte[] ToPng()
    {
        using var bitmap = new SKBitmap(Width, Height);
        for (var y = 0; y < Height; y++)
        {
            for (var x = 0; x < Width; x++)
            {
                bitmap.SetPixel(x, y, IsWhite(x, y) ? SKColors.White : SKColors.Black);
            }
        }

        using var image = SKImage.FromBitmap(bitmap);
        using var data = image.Encode(SKEncodedImageFormat.Png, 100);
        return data.ToArray();
    }

    /// <inheritdoc />
    public bool Equals(MonochromeFrame? other) =>
        other is not null && Width == other.Width && Height == other.Height && _data.AsSpan().SequenceEqual(other._data);

    /// <inheritdoc />
    public override bool Equals(object? obj) => Equals(obj as MonochromeFrame);

    /// <inheritdoc />
    public override int GetHashCode()
    {
        var hash = new HashCode();
        hash.Add(Width);
        hash.Add(Height);
        hash.AddBytes(_data);
        return hash.ToHashCode();
    }

    private static MonochromeFrame CreateWhite(int width, int height)
    {
        var data = new byte[(width + 7) / 8 * height];
        data.AsSpan().Fill(0xFF);
        return new MonochromeFrame(width, height, data);
    }

    private void SetBlack(int x, int y) => _data[y * BytesPerRow + x / 8] &= (byte)~(0x80 >> (x % 8));
}
```

```csharp
using Namotion.Interceptor;
using Namotion.Interceptor.Registry;
using Namotion.Interceptor.Registry.Abstractions;
using Namotion.Interceptor.Registry.Paths;
using Namotion.Interceptor.Tracking.Parent;

namespace Namotion.Devices.Displays.Templates;

/// <summary>
/// Resolves registry paths, such as <c>Devices[gpio]</c>, from the root of the graph that holds a subject, for contexts
/// without a path resolver of their host.
/// </summary>
internal static class SubjectGraph
{
    // Bounds the walk, so that a cycle of parents cannot hang it.
    private const int MaximumDepth = 64;

    public static IInterceptorSubject GetRoot(IInterceptorSubject subject)
    {
        for (var depth = 0; depth < MaximumDepth; depth++)
        {
            var parents = subject.GetParents();
            if (parents.Length == 0)
            {
                return subject;
            }

            subject = parents[0].Property.Subject;
        }

        return subject;
    }

    public static RegisteredSubjectProperty? TryGetProperty(IInterceptorSubject from, string path) =>
        GetRoot(from).TryGetRegisteredSubject() is { } root &&
        DefaultPathProvider.Instance.TryGetPropertyFromPath(root, path) is { Index: null, Property: var property }
            ? property
            : null;

    public static IInterceptorSubject? TryGetSubject(IInterceptorSubject from, string path) =>
        GetRoot(from).TryGetRegisteredSubject() is { } root
            ? DefaultPathProvider.Instance.TryGetSubjectFromPath(root, path)?.Subject
            : null;
}
```

```csharp
using System.Collections.Immutable;
using System.Diagnostics.CodeAnalysis;
using System.Globalization;
using HomeBlaze.Abstractions;
using HomeBlaze.Abstractions.Attributes;
using Namotion.Interceptor;
using Namotion.Interceptor.Expressions;
using Namotion.Interceptor.Registry.Abstractions;
using Namotion.Interceptor.Registry.Paths;

namespace Namotion.Devices.Displays.Templates;

/// <summary>
/// The template renderer of a context without one of its host: paths in the syntax of the registry resolve from the root
/// of the graph, values are written in the invariant culture, secret configuration values are left out, and no pipe is
/// known.
/// </summary>
internal sealed class RegistryTextTemplateRenderer : ITextTemplateRenderer
{
    public static RegistryTextTemplateRenderer Instance { get; } = new();

    public bool TryParse(string text, [NotNullWhen(true)] out ITextTemplate? template, [NotNullWhen(false)] out string? error)
    {
        ArgumentNullException.ThrowIfNull(text);
        if (!TextTemplate.TryParse(text, DefaultPathProvider.Instance.PathSeparator, ImmutableArray<string>.Empty, out var parsed, out error))
        {
            template = null;
            return false;
        }

        foreach (var expression in parsed.Expressions)
        {
            if (expression.Pipes.Length > 0)
            {
                template = null;
                error = $"Unknown pipe '{expression.Pipes[0].Name}'";
                return false;
            }
        }

        template = new RegistryTextTemplate(parsed);
        return true;
    }

    private sealed class RegistryTextTemplate(TextTemplate template) : ITextTemplate
    {
        public string Text => template.Text;

        public string Render(IInterceptorSubject scope, Func<string, string>? encode = null)
        {
            ArgumentNullException.ThrowIfNull(scope);
            var context = new RegistryExpressionContext(scope, encode);
            return template.Render(ref context);
        }
    }

    private readonly struct RegistryExpressionContext(IInterceptorSubject scope, Func<string, string>? encode) : IExpressionContext
    {
        public ExpressionValue ResolvePath(string path) =>
            SubjectGraph.TryGetProperty(scope, path) is { } property && !IsSecret(property)
                ? ExpressionValue.FromProperty(property, property.GetValue())
                : ExpressionValue.Missing;

        // TryParse declares no variable.
        public ExpressionValue ResolveVariable(string variable, string? memberPath) => ExpressionValue.Missing;

        // TryParse refuses pipes.
        public ExpressionValue ApplyPipe(string name, ImmutableArray<string> arguments, ExpressionValue input) => input;

        public string ToText(ExpressionValue result)
        {
            var text = result.Value switch
            {
                null => string.Empty,
                IFormattable formattable => formattable.ToString(null, CultureInfo.InvariantCulture),
                var value => value.ToString() ?? string.Empty
            };
            return encode is null ? text : encode(text);
        }

        private static bool IsSecret(RegisteredSubjectProperty property) =>
            property.ReflectionAttributes.Any(attribute => attribute is ConfigurationAttribute { IsSecret: true });
    }
}
```

```csharp
using System.Diagnostics.CodeAnalysis;
using System.Security;
using System.Xml;
using HomeBlaze.Abstractions;
using Namotion.Interceptor;
using SkiaSharp;
using Svg.Skia;
using Svg.Skia.TypefaceProviders;

namespace Namotion.Devices.Displays.Templates;

/// <summary>
/// An SVG template with expressions, parsed once and rendered to a frame of a given size. Thread-safe.
/// </summary>
internal sealed class DisplayTemplate
{
    private static readonly Func<string, string> EscapeXml = value => SecurityElement.Escape(value) ?? string.Empty;

    private readonly ITextTemplate _template;

    private DisplayTemplate(ITextTemplate template)
    {
        _template = template;
    }

    public static bool TryParse(
        ITextTemplateRenderer renderer,
        string text,
        [NotNullWhen(true)] out DisplayTemplate? template,
        [NotNullWhen(false)] out string? error)
    {
        if (!renderer.TryParse(text, out var parsed, out error))
        {
            template = null;
            return false;
        }

        template = new DisplayTemplate(parsed);
        return true;
    }

    /// <summary>
    /// Evaluates the expressions with XML-escaped values, draws the SVG scaled to fit on white and converts the result.
    /// </summary>
    /// <exception cref="DisplayTemplateException">The SVG cannot be loaded.</exception>
    public MonochromeFrame Render(IInterceptorSubject scope, int width, int height)
    {
        var svgText = _template.Render(scope, EscapeXml);

        using var svg = new SKSvg();
        svg.Settings.EnableJavaScript = false;
        svg.Settings.EnableExternalJavaScript = false;
        svg.Settings.TypefaceProviders = [new FontManagerTypefaceProvider(), EmbeddedTypefaceProvider.Instance];

        SKPicture? picture;
        try
        {
            picture = svg.FromSvg(svgText);
        }
        catch (XmlException exception)
        {
            throw new DisplayTemplateException($"SVG: {exception.Message}", exception);
        }

        if (picture is null)
        {
            throw new DisplayTemplateException("SVG: the document has no content.", null);
        }

        using var bitmap = new SKBitmap(width, height);
        using (var canvas = new SKCanvas(bitmap))
        {
            canvas.Clear(SKColors.White);
            var bounds = picture.CullRect;
            if (bounds.Width > 0 && bounds.Height > 0)
            {
                var scale = Math.Min(width / bounds.Width, height / bounds.Height);
                canvas.Scale(scale);
                canvas.Translate(-bounds.Left, -bounds.Top);
            }

            canvas.DrawPicture(picture);
        }

        return MonochromeFrame.FromBitmap(bitmap);
    }
}

/// A display template that rendered to an SVG that cannot be loaded.
/// </summary>
public sealed class DisplayTemplateException(string message, Exception? innerException) : Exception(message, innerException);
```

```csharp
namespace Namotion.Devices.Displays;

/// <summary>
/// Decides when a display renders: every interval after the last render, and on a change as soon as an interval has
/// passed since the last frame was pushed. Only <see cref="MarkDirty"/> is called from other threads.
/// </summary>
internal sealed class DisplayRefreshScheduler
{
    private int _isDirty;
    private DateTimeOffset? _lastRender;
    private DateTimeOffset? _lastPush;

    public bool IsDirty => Volatile.Read(ref _isDirty) == 1;

    public void MarkDirty() => Volatile.Write(ref _isDirty, 1);

    /// <summary>Called right before a render, so that a change during the render marks the display dirty again.</summary>
    public void OnRendering() => Volatile.Write(ref _isDirty, 0);

    public void OnRendered(DateTimeOffset now, bool isPushed)
    {
        _lastRender = now;
        if (isPushed)
        {
            _lastPush = now;
        }
    }

    /// <summary>Gets the time until the next render, zero when it is due.</summary>
    public TimeSpan GetDelay(DateTimeOffset now, TimeSpan interval)
    {
        if (_lastRender is not { } lastRender)
        {
            return TimeSpan.Zero;
        }

        var untilTimer = lastRender + interval - now;
        if (IsDirty)
        {
            var untilChange = _lastPush is { } lastPush ? lastPush + interval - now : TimeSpan.Zero;
            untilTimer = untilChange < untilTimer ? untilChange : untilTimer;
        }

        return untilTimer > TimeSpan.Zero ? untilTimer : TimeSpan.Zero;
    }
}
```

```csharp
using HomeBlaze.Abstractions;
using HomeBlaze.Abstractions.Attributes;
using Microsoft.Extensions.Hosting;
using Namotion.Devices.Displays.Templates;
using Namotion.Interceptor;
using Namotion.Interceptor.Attributes;
using Namotion.Interceptor.Tracking.Change;
using Namotion.Interceptor.Tracking.Recorder;

namespace Namotion.Devices.Displays;

/// <summary>
/// A display that shows an SVG template with expressions, rendered whenever a value it shows changes and on a timer,
/// and pushed to the hardware no more often than <see cref="RefreshInterval"/> and only when the pixels changed.
/// Subclasses connect to the hardware and show frames.
/// </summary>
[InterceptorSubject]
public abstract partial class PixelDisplay : BackgroundService, IConfigurable, IMonitoredService, ITitleProvider, IIconProvider
{
    private readonly TimeProvider _timeProvider;
    private readonly DisplayRefreshScheduler _scheduler = new();
    private readonly SemaphoreSlim _signal = new(0, 1);
    private readonly PropertyChangeCallback _onWatchedPropertyChanged;

    // Owned by the service loop.
    private readonly Dictionary<PropertyReference, IDisposable> _subscriptions = new();
    private MonochromeFrame? _shownFrame;

    // Replaced as a whole, so that the loop and a preview see a template and its error together.
    private volatile ParsedTemplate _template = new(null, null, null);

    private volatile MonochromeFrame? _shownLogicalFrame;
    private volatile string? _reconnectMessage;
    private int _isReconnectRequested;

    /// <summary>Creates a display that measures time with <paramref name="timeProvider"/>.</summary>
    protected PixelDisplay(TimeProvider timeProvider)
    {
        _timeProvider = timeProvider;
        _onWatchedPropertyChanged = OnWatchedPropertyChanged;
        Template = string.Empty;
        RefreshInterval = TimeSpan.FromSeconds(60);
        RetryInterval = TimeSpan.FromSeconds(30);
        Status = ServiceStatus.Stopped;
    }

    /// <summary>The name shown as the title.</summary>
    [Configuration]
    public partial string? Name { get; set; }

    /// <summary>
    /// The SVG to show, with expressions between double braces whose values are XML-escaped, such as
    /// <c>&lt;text x="4" y="40"&gt;{{ Devices/Wallbox/ChargingPower }}&lt;/text&gt;</c>. It is scaled to fit the display.
    /// </summary>
    [Configuration]
    public partial string Template { get; set; }

    /// <summary>The shortest time between two frames pushed to the display, and the interval of the timer render.</summary>
    [Configuration]
    public partial TimeSpan RefreshInterval { get; set; }

    /// <summary>The time to wait before connecting again after a failure.</summary>
    [Configuration]
    public partial TimeSpan RetryInterval { get; set; }

    /// <inheritdoc />
    [State(IsDiscrete = true)]
    public partial ServiceStatus Status { get; protected set; }

    /// <inheritdoc />
    [State]
    public partial string? StatusMessage { get; protected set; }

    /// <summary>When a frame was last pushed to the display.</summary>
    [State]
    public partial DateTimeOffset? LastRefreshTime { get; protected set; }

    /// <inheritdoc />
    public string? Title => string.IsNullOrWhiteSpace(Name) ? null : Name;

    /// <inheritdoc />
    public string IconName => "Tv";

    /// <inheritdoc />
    [Derived]
    public string? IconColor => Status switch
    {
        ServiceStatus.Running => "Success",
        ServiceStatus.Error => "Error",
        _ => null
    };

    /// <summary>The size, orientation and pixel format of the display.</summary>
    protected abstract DisplayGeometry Geometry { get; }

    /// <summary>Connects to the hardware. Throw <see cref="DisplayUnavailableException"/> when a dependency is not there yet.</summary>
    protected internal abstract Task ConnectAsync(CancellationToken cancellationToken);

    /// <summary>Shows a frame in the native orientation and size of <see cref="Geometry"/>.</summary>
    protected internal abstract Task ShowAsync(MonochromeFrame frame, CancellationToken cancellationToken);

    /// <summary>Disconnects from the hardware. Called after a failed connect too; must not throw for a partial connection.</summary>
    protected internal abstract Task DisconnectAsync(CancellationToken cancellationToken);

    /// <summary>
    /// Asks the service loop to disconnect and connect again, such as when a dependency went away. The display shows
    /// <see cref="ServiceStatus.Unavailable"/> with <paramref name="message"/> until it connected. Thread-safe.
    /// </summary>
    protected void RequestReconnect(string message)
    {
        _reconnectMessage = message;
        Volatile.Write(ref _isReconnectRequested, 1);
        Signal();
    }

    /// <summary>Gets the last frame pushed to the display as a PNG in the orientation of the template, or null.</summary>
    public byte[]? GetShownFramePng() => _shownLogicalFrame?.ToPng();

    /// <summary>Renders a template with the current values without showing it, such as for an editor.</summary>
    public DisplayPreview RenderPreview(string template)
    {
        ArgumentNullException.ThrowIfNull(template);
        var parsed = Parse(template);
        if (parsed.Template is null)
        {
            return new DisplayPreview(null, parsed.Error);
        }

        try
        {
            var geometry = Geometry;
            return new DisplayPreview(parsed.Template.Render(this, geometry.LogicalWidth, geometry.LogicalHeight).ToPng(), null);
        }
        catch (DisplayTemplateException exception)
        {
            return new DisplayPreview(null, exception.Message);
        }
    }

    /// <inheritdoc />
    public Task ApplyConfigurationAsync(CancellationToken cancellationToken)
    {
        _template = Parse(Template);
        _scheduler.MarkDirty();
        RequestReconnect("Applying the configuration...");
        return Task.CompletedTask;
    }

    /// <inheritdoc />
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _template = Parse(Template);
        try
        {
            while (!stoppingToken.IsCancellationRequested)
            {
                try
                {
                    Volatile.Write(ref _isReconnectRequested, 0);
                    Status = ServiceStatus.Starting;
                    StatusMessage = _reconnectMessage ?? "Connecting...";
                    await ConnectAsync(stoppingToken);
                    _shownFrame = null;
                    _reconnectMessage = null;
                    Status = ServiceStatus.Running;
                    StatusMessage = _template.Error;
                    await RunAsync(stoppingToken);
                }
                catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
                {
                    break;
                }
                catch (Exception exception)
                {
                    Status = exception is DisplayUnavailableException ? ServiceStatus.Unavailable : ServiceStatus.Error;
                    StatusMessage = exception.Message;
                    await DisconnectQuietlyAsync();
                    await WaitAsync(RetryInterval, stoppingToken);
                    continue;
                }

                // A reconnect was requested: the next connect shows its message, and a connect that throws
                // DisplayUnavailableException shows Unavailable.
                await DisconnectQuietlyAsync();
            }
        }
        catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
        {
            // The service stops.
        }
        finally
        {
            ClearSubscriptions();
            await DisconnectQuietlyAsync();
            StatusMessage = null;
            Status = ServiceStatus.Stopped;
        }
    }

    /// <summary>The number of renders since the start, for tests.</summary>
    internal int RenderCount { get; private set; }

    private async Task RunAsync(CancellationToken stoppingToken)
    {
        while (Volatile.Read(ref _isReconnectRequested) == 0)
        {
            var now = _timeProvider.GetUtcNow();
            var interval = RefreshInterval;
            if (_scheduler.GetDelay(now, interval) == TimeSpan.Zero)
            {
                _scheduler.OnRendering();
                var isPushed = await RenderAndShowAsync(now, stoppingToken);
                _scheduler.OnRendered(now, isPushed);
            }

            await WaitAsync(_scheduler.GetDelay(_timeProvider.GetUtcNow(), interval), stoppingToken);
        }
    }

    private async Task<bool> RenderAndShowAsync(DateTimeOffset now, CancellationToken stoppingToken)
    {
        var template = _template;
        if (template.Template is null)
        {
            StatusMessage = template.Error;
            return false;
        }

        var geometry = Geometry;
        MonochromeFrame frame;
        var recorder = ReadPropertyRecorder.Start();
        try
        {
            frame = template.Template.Render(this, geometry.LogicalWidth, geometry.LogicalHeight);
        }
        catch (DisplayTemplateException exception)
        {
            recorder.GetPropertiesAndDispose();
            StatusMessage = exception.Message;
            return false;
        }
        finally
        {
            RenderCount++;
        }

        UpdateSubscriptions(recorder.GetPropertiesAndDispose());
        StatusMessage = null;

        if (frame.Equals(_shownLogicalFrame) && _shownFrame is not null)
        {
            return false;
        }

        var nativeFrame = frame.Rotate(geometry.Rotation);
        await ShowAsync(nativeFrame, stoppingToken);
        _shownFrame = nativeFrame;
        _shownLogicalFrame = frame;
        LastRefreshTime = now;
        return true;
    }

    private ParsedTemplate Parse(string text)
    {
        var renderer = ((IInterceptorSubject)this).Context.TryGetService<ITextTemplateRenderer>() ?? RegistryTextTemplateRenderer.Instance;
        return DisplayTemplate.TryParse(renderer, text, out var template, out var error)
            ? new ParsedTemplate(text, template, null)
            : new ParsedTemplate(text, null, $"Template: {error}");
    }

    private void UpdateSubscriptions(ICollection<PropertyReference> reads)
    {
        foreach (var property in _subscriptions.Keys.Where(property => !reads.Contains(property)).ToArray())
        {
            _subscriptions.Remove(property, out var subscription);
            subscription!.Dispose();
        }

        foreach (var property in reads)
        {
            if (_subscriptions.ContainsKey(property))
            {
                continue;
            }

            try
            {
                _subscriptions[property] = property.SubscribeInline(_onWatchedPropertyChanged);
            }
            catch (ArgumentException)
            {
                // Not intercepted nor derived, so it never notifies: the timer render shows its changes.
            }
        }
    }

    private void ClearSubscriptions()
    {
        foreach (var subscription in _subscriptions.Values)
        {
            subscription.Dispose();
        }

        _subscriptions.Clear();
    }

    // Runs on the writing thread: only flags and signals.
    private void OnWatchedPropertyChanged(in SubjectPropertyChange change)
    {
        _scheduler.MarkDirty();
        Signal();
    }

    private void Signal()
    {
        try
        {
            if (_signal.CurrentCount == 0)
            {
                _signal.Release();
            }
        }
        catch (SemaphoreFullException)
        {
            // Another thread signaled first.
        }
    }

    private async Task WaitAsync(TimeSpan delay, CancellationToken stoppingToken)
    {
        using var timeout = CancellationTokenSource.CreateLinkedTokenSource(stoppingToken);
        var delayTask = Task.Delay(delay, _timeProvider, timeout.Token);
        var signalTask = _signal.WaitAsync(timeout.Token);
        await Task.WhenAny(delayTask, signalTask);
        await timeout.CancelAsync();
        stoppingToken.ThrowIfCancellationRequested();
    }

    private async Task DisconnectQuietlyAsync()
    {
        try
        {
            await DisconnectAsync(CancellationToken.None);
        }
        catch
        {
            // A disconnect that fails leaves nothing to release: the next connect starts over.
        }
    }

    private sealed record ParsedTemplate(string? Text, DisplayTemplate? Template, string? Error);
}
```

Protocol, from Waveshare's `epd2in13_V4.py` (V1.0, 2023-06-25) and `epdconfig.py`:
- Command: DC low, write the byte. Data: DC high, write the bytes. Chip select comes from the SPI hardware (spidev 0.0 = CE0).
- BUSY high means busy; poll every 10 ms.
- Reset: RST high, 20 ms, low, 2 ms, high, 20 ms. Power (PWR, GPIO 18) high before init, low after sleep.
- Init: reset, wait busy, `0x12` (software reset), wait busy, `0x01` [(H-1) & 0xFF, (H-1) >> 8, 0x00], `0x11` [0x03], window, cursor, `0x3C` [0x05], `0x21` [0x00, 0x80], `0x18` [0x80], wait busy.
- Window: `0x44` [0x00, (W-1) >> 3], `0x45` [0x00, 0x00, (H-1) & 0xFF, (H-1) >> 8]. Cursor: `0x4E` [0x00], `0x4F` [0x00, 0x00].
- Full refresh with base image for later partial refreshes: `0x24` frame, `0x26` frame, `0x22` [0xF7], `0x20`, wait busy.
- Partial refresh: RST low, 1 ms, high; `0x3C` [0x80], `0x01` [...], `0x11` [0x03], window, cursor, `0x24` frame, `0x22` [0xFF], `0x20`, wait busy.
- Sleep: `0x10` [0x01], 2 s, then RST, DC and PWR low.
- SPI: 4 MHz, mode 0. Write the frame in chunks of at most 4096 bytes (the default spidev buffer size).

```csharp
using System.Device.Spi;

namespace Namotion.Devices.Displays.Tests.Ssd1680;

/// <summary>Records each write as a command (DC low) or data (DC high), as the pins report the DC level.</summary>
internal sealed class RecordingSpiDevice(FakeDisplayPins pins) : SpiDevice
{
    public List<(bool IsData, byte[] Bytes)> Writes { get; } = [];

    public bool IsDisposed { get; private set; }

    public override SpiConnectionSettings ConnectionSettings { get; } = new(0, 0);

    public IEnumerable<byte> Commands => Writes.Where(write => !write.IsData).Select(write => write.Bytes[0]);

    public byte[] DataAfter(byte command, int occurrence = 0)
    {
        var index = Writes.Select((write, position) => (write, position))
            .Where(entry => !entry.write.IsData && entry.write.Bytes[0] == command)
            .Skip(occurrence).First().position;
        return Writes.Skip(index + 1).TakeWhile(write => write.IsData).SelectMany(write => write.Bytes).ToArray();
    }

    public override void Write(ReadOnlySpan<byte> buffer) => Writes.Add((pins.IsDataMode, buffer.ToArray()));

    public override void WriteByte(byte value) => Writes.Add((pins.IsDataMode, [value]));

    public override byte ReadByte() => throw new NotSupportedException();

    public override void Read(Span<byte> buffer) => throw new NotSupportedException();

    public override void TransferFullDuplex(ReadOnlySpan<byte> writeBuffer, Span<byte> readBuffer) => throw new NotSupportedException();

    protected override void Dispose(bool disposing)
    {
        IsDisposed = true;
        base.Dispose(disposing);
    }
}
```

```csharp
    private readonly Dictionary<int, PinValue> _levels = new();

    public List<(int PinNumber, PinValue Value)> PinWrites { get; } = [];

    /// <summary>The number of BUSY reads that report busy before it reports idle; -1 for always busy.</summary>
    public int BusyReads { get; set; }

    public bool IsDataMode => _levels.GetValueOrDefault(pins.DataCommandPin) == PinValue.High;

    public void Write(int pinNumber, PinValue value)
    {
        _levels[pinNumber] = value;
        PinWrites.Add((pinNumber, value));
    }

    public PinValue Read(int pinNumber)
    {
        if (pinNumber != pins.BusyPin)
        {
            return _levels.GetValueOrDefault(pinNumber);
        }

        if (BusyReads == 0)
        {
            return PinValue.Low;
        }

        if (BusyReads > 0)
        {
            BusyReads--;
        }

        return PinValue.High;
    }
```

```csharp
using System.Device.Gpio;
using System.Device.Spi;

namespace Namotion.Devices.Displays.Ssd1680;

/// <summary>
/// The command protocol of an SSD1680 e-paper controller in black and white, after the reference driver of the
/// Waveshare 2.13 inch e-Paper HAT V4. Not thread-safe: one display loop uses it.
/// </summary>
internal sealed class Ssd1680Controller(
    SpiDevice spi,
    IDisplayPins pins,
    Ssd1680Pins pinNumbers,
    int width,
    int height,
    Func<TimeSpan, CancellationToken, Task> delay)
{
    private const int SpiChunkSize = 4096;
    private static readonly TimeSpan BusyPollInterval = TimeSpan.FromMilliseconds(10);
    private const int MaximumBusyPolls = 1000;

    public async Task InitializeAsync(CancellationToken cancellationToken)
    {
        if (pinNumbers.PowerPin is { } powerPin)
        {
            pins.Write(powerPin, PinValue.High);
        }

        pins.Write(pinNumbers.ResetPin, PinValue.High);
        await delay(TimeSpan.FromMilliseconds(20), cancellationToken);
        pins.Write(pinNumbers.ResetPin, PinValue.Low);
        await delay(TimeSpan.FromMilliseconds(2), cancellationToken);
        pins.Write(pinNumbers.ResetPin, PinValue.High);
        await delay(TimeSpan.FromMilliseconds(20), cancellationToken);

        await WaitUntilIdleAsync(cancellationToken);
        SendCommand(0x12);
        await WaitUntilIdleAsync(cancellationToken);

        SendDriverOutputAndAddressing();
        SendCommand(0x3C, 0x05);
        SendCommand(0x21, 0x00, 0x80);
        SendCommand(0x18, 0x80);
        await WaitUntilIdleAsync(cancellationToken);
    }

    /// <summary>A full refresh, which also stores the frame as the base image of later partial refreshes.</summary>
    public async Task ShowFullAsync(MonochromeFrame frame, CancellationToken cancellationToken)
    {
        EnsureSize(frame);
        SendCommand(0x24);
        SendData(frame.Data.Span);
        SendCommand(0x26);
        SendData(frame.Data.Span);
        SendCommand(0x22, 0xF7);
        SendCommand(0x20);
        await WaitUntilIdleAsync(cancellationToken);
    }

    public async Task ShowPartialAsync(MonochromeFrame frame, CancellationToken cancellationToken)
    {
        EnsureSize(frame);
        pins.Write(pinNumbers.ResetPin, PinValue.Low);
        await delay(TimeSpan.FromMilliseconds(1), cancellationToken);
        pins.Write(pinNumbers.ResetPin, PinValue.High);

        SendCommand(0x3C, 0x80);
        SendDriverOutputAndAddressing();
        SendCommand(0x24);
        SendData(frame.Data.Span);
        SendCommand(0x22, 0xFF);
        SendCommand(0x20);
        await WaitUntilIdleAsync(cancellationToken);
    }

    /// <summary>Enters deep sleep and powers the panel off; the panel keeps its image.</summary>
    public async Task SleepAsync(CancellationToken cancellationToken)
    {
        SendCommand(0x10, 0x01);
        await delay(TimeSpan.FromSeconds(2), cancellationToken);
        pins.Write(pinNumbers.ResetPin, PinValue.Low);
        pins.Write(pinNumbers.DataCommandPin, PinValue.Low);
        if (pinNumbers.PowerPin is { } powerPin)
        {
            pins.Write(powerPin, PinValue.Low);
        }
    }

    private void SendDriverOutputAndAddressing()
    {
        var lastRow = height - 1;
        SendCommand(0x01, (byte)(lastRow & 0xFF), (byte)(lastRow >> 8), 0x00);
        SendCommand(0x11, 0x03);
        SendCommand(0x44, 0x00, (byte)((width - 1) >> 3));
        SendCommand(0x45, 0x00, 0x00, (byte)(lastRow & 0xFF), (byte)(lastRow >> 8));
        SendCommand(0x4E, 0x00);
        SendCommand(0x4F, 0x00, 0x00);
    }

    private void SendCommand(byte command, params ReadOnlySpan<byte> data)
    {
        pins.Write(pinNumbers.DataCommandPin, PinValue.Low);
        spi.WriteByte(command);
        if (data.Length > 0)
        {
            SendData(data);
        }
    }

    private void SendData(ReadOnlySpan<byte> data)
    {
        pins.Write(pinNumbers.DataCommandPin, PinValue.High);
        for (var offset = 0; offset < data.Length; offset += SpiChunkSize)
        {
            spi.Write(data.Slice(offset, Math.Min(SpiChunkSize, data.Length - offset)));
        }
    }

    private async Task WaitUntilIdleAsync(CancellationToken cancellationToken)
    {
        for (var poll = 0; poll < MaximumBusyPolls; poll++)
        {
            if (pins.Read(pinNumbers.BusyPin) == PinValue.Low)
            {
                return;
            }

            await delay(BusyPollInterval, cancellationToken);
        }

        throw new TimeoutException("The e-paper display stayed busy.");
    }

    private void EnsureSize(MonochromeFrame frame)
    {
        if (frame.Width != width || frame.Height != height)
        {
            throw new ArgumentException($"The frame is {frame.Width}x{frame.Height}, the display is {width}x{height}.", nameof(frame));
        }
    }
}
```

```csharp
        }

        var pins = new Ssd1680Pins(ResetPin, DataCommandPin, BusyPin, PowerPin);
        _lease = gpio.LeasePins(this, CreateLeaseRequest(pins));
        _lease.Revoked.Register(() => RequestReconnect("GPIO not running"));

        _spiDevice = _createSpiDevice(new SpiConnectionSettings(SpiBus, SpiChipSelect)
        {
            ClockFrequency = 4_000_000,
            Mode = SpiMode.Mode0
        });

        _controller = new Ssd1680Controller(_spiDevice, new LeasedDisplayPins(_lease), pins, Width, Height, _delay);
        await _controller.InitializeAsync(cancellationToken);
        _partialRefreshCount = 0;
        _lastFullRefreshTime = null;
    }

    /// <inheritdoc />
    protected internal override async Task ShowAsync(MonochromeFrame frame, CancellationToken cancellationToken)
    {
        var controller = _controller ?? throw new InvalidOperationException("The display is not connected.");
        var now = _timeProvider.GetUtcNow();
        if (_lastFullRefreshTime is not { } lastFullRefreshTime ||
            _partialRefreshCount >= MaximumPartialRefreshes ||
            now - lastFullRefreshTime >= MaximumTimeBetweenFullRefreshes)
        {
            await controller.ShowFullAsync(frame, cancellationToken);
            _partialRefreshCount = 0;
            _lastFullRefreshTime = now;
            return;
        }

        await controller.ShowPartialAsync(frame, cancellationToken);
        _partialRefreshCount++;
    }

    /// <inheritdoc />
    protected internal override async Task DisconnectAsync(CancellationToken cancellationToken)
    {
        try
        {
            if (_controller is not null && _lease is { Revoked.IsCancellationRequested: false })
            {
                await _controller.SleepAsync(cancellationToken);
            }
        }
        finally
        {
            _controller = null;
            _spiDevice?.Dispose();
            _spiDevice = null;
            _lease?.Dispose();
            _lease = null;
        }
    }

    private GpioSubject? ResolveGpio()
    {
        if (string.IsNullOrWhiteSpace(Gpio))
        {
            return null;
        }

        var subject = (IInterceptorSubject)this;
        var resolver = subject.Context.TryGetService<ISubjectPathResolver>();
        return (resolver is not null
            ? resolver.ResolveSubject(Gpio, PathStyle.Canonical, this)
            : SubjectGraph.TryGetSubject(this, Gpio)) as GpioSubject;
    }

    private GpioPinLeaseRequest[] CreateLeaseRequest(Ssd1680Pins pins)
    {
        var requests = new List<GpioPinLeaseRequest>
        {
            new(pins.ResetPin, GpioPinLeaseMode.Output),
            new(pins.DataCommandPin, GpioPinLeaseMode.Output),
            new(pins.BusyPin, GpioPinLeaseMode.Input)
        };

        if (pins.PowerPin is { } powerPin)
        {
            requests.Add(new(powerPin, GpioPinLeaseMode.Output));
        }

        // The SPI0 pins of a Raspberry Pi, which the kernel driver owns: GPIO must not open them.
        if (SpiBus == 0)
        {
            requests.Add(new(9, GpioPinLeaseMode.Reserved));
            requests.Add(new(10, GpioPinLeaseMode.Reserved));
            requests.Add(new(11, GpioPinLeaseMode.Reserved));
            requests.Add(new(SpiChipSelect == 0 ? 8 : 7, GpioPinLeaseMode.Reserved));
        }

        return requests.ToArray();
    }
}
```

Note on `Revoked.Register`: the registration is never disposed; it ends with the lease, which is never reused.
